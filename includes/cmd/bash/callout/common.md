```sh
# Arithmetic expansion
$((...))

# Alias for test
[ ... ]

# Command substitution
$( ... )

# Variable substitution
${ ... }

# String substitution (first match only)
${STRING/SEARCH/REPLACE}

# Global string substition (all matches)
${STRING//SEARCH/REPLACE}

# Variable slicing
${VAR:START:SIZE}

# Brace expansion
filename{,.new}
```
